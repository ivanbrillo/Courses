class InvertedIndex:

    class PostingListIterator:
        def __init__(self, docids, freqs, doc):
            self.docids = docids
            self.freqs = freqs
            self.pos = 0
            self.doc = doc
            
        def docid(self):
            if self.is_end_list():
                return math.inf
            return self.docids[self.pos]

        def score(self):
            if self.is_end_list():
                return math.inf
            return self.freqs[self.pos]/self.doc[self.docid()][1]

        def next(self, target=None):
            if not target:
                if not self.is_end_list():
                    self.pos += 1
            else:
                if target > self.docid():
                    try:
                        self.pos = self.docids.index(target, self.pos)
                    except ValueError:
                        self.pos = len(self.docids)

        def is_end_list(self):
            return self.pos == len(self.docids)


        def len(self):
            return len(self.docids)
            
            
    def __init__(self, lex, inv, doc, stats):
        self.lexicon = lex
        self.inv = inv
        self.doc = doc
        self.stat = stats

    def num_docs(self):
        return self.stats['num_docs']

    def get_posting(self, termid):
        return InvertedIndex.PostingListIterator(self.inv['docids'][termid], self.inv['freqs'][termid], self.doc)

    def get_termids(self, tokens):
        return [self.lexicon[token][0] for token in tokens if token in self.lexicon]
    
    def get_postings(self, termids):
        return [self.get_posting(termid) for termid in termids]

def boolean_and(postings):  # query terms posting
    results = []
    # We sort the posting lists from the shortest to the longest
    postings = sorted(postings, key=lambda p: p.len())
    # We scan sequentially through the shortest posting list only
    current_docid = postings[0].docid()
    while current_docid != math.inf:
        found = True
        # We look for the current docid is all remaining posting lists
        for posting in postings[1:]:
            posting.next(current_docid)
            if posting.docid() != current_docid:
                found = False
                break
        # If the current docid is in all posting lists, we add it to results
        if found:
            results.append(current_docid)
        # We move forward in the shortest posting list
        postings[0].next()
        current_docid = postings[0].docid()
    return results

class TopQueue:
    def __init__(self, k=10, threshold=0.0):
        self.queue = []
        self.k = k
        self.threshold = threshold

    def size(self):
        return len(self.queue)

    def would_enter(self, score):
        return score > self.threshold

    def clear(self, new_threshold=None):
        self.queue = []
        if new_threshold:
            self.threshold = new_threshold

    def __repr__(self):
        return f'<{self.size()} items, th={self.threshold} {self.queue}' 
        
    def insert(self, docid, score):
        if score > self.threshold:
            if self.size() >= self.k:
                heapq.heapreplace(self.queue, (score, docid))
            else:
                heapq.heappush(self.queue, (score, docid))
            if self.size() >= self.k:
                self.threshold = max(self.threshold, self.queue[0][0])
            return True
        return False

def taat(postings, k=10):
    A = defaultdict(float)     # accumulator per document
    for posting in postings:
        current_docid = posting.docid()
        while current_docid != math.inf:
            A[current_docid] += posting.score()
            posting.next()
            current_docid = posting.docid()
    top = TopQueue(k)
    for docid, score in A.items():
        top.insert(docid, score)
    return sorted(top.queue, reverse=True)
    
  def daat(postings, k=10):
    top = TopQueue(k)
    current_docid = min_docid(postings)
    while current_docid != math.inf:
        score = 0       # score per current doc
        next_docid = math.inf
        for posting in postings:
            if posting.docid() == current_docid:
                score += posting.score()
                posting.next()
            if not posting.is_end_list():
                next_docid = min(next_docid, posting.docid())
        top.insert(current_docid, score)
        current_docid = next_docid
    return sorted(top.queue, reverse=True)

Evaluation

  • Offline evaluation: the IR system is evaluated without any interaction with users. We assume to have assessors, that can correctly tell us what is relevant and what is not.
    • based on experimental collections C=(D,T,A)\mathcal{C} = (D,T,A)C=(D,T,A)﻿ where:
      • DDD﻿ is a a corpus of documents in the domain of interest
        • docno and body
      • TTT﻿ is a set of topics representing (a surrogate of) the user information needs
        • ID, title (search engine query), description, narrative (instruction for assessors on when to consider a doc relevant)
      • AAA﻿ is a list of human-made relevance assessments (binary or graded)
        • topic id, docno, judgment, iteration (unused and equal to ‘0’)
    • Tuple (t,d,p)∈C(t, d, p) \in \mathcal{C}(t,d,p)∈C﻿
  • Online evaluation: the IR system is evaluated while interacting with the user, telling us what is relevant and what is not.

A run [ranked list from the engine] is a textual file whose fields (all mandatory) are separated by a tab or a space character. The fields of a run are the following:

  • a topic id, the topic id as in $T$,
  • fixed, unused and always equal to the string 'Q0',
  • a docno, the document number as in $D$,
  • a rank, the position of the document in the ranked list for the topic,
  • a score, the score given by the IR system to the document for the topic (a real number), and
  • a run id, an identifier, equal for all the lines in the run, of the IR system search.

The relevance assessments AAA﻿ are then used to judge each of the retrieved documents and to produce the assessed run.

  • Relevance labels in the assessed run are then mapped to numbers, called relevance weights and this originates the weighted assessed run.
  • Typically, 0 and 1 are used in the case of binary relevance labels and integer numbers in the case of graded relevance labels.

The weighted assessed run is finally scored using one of the many evaluation measures M\mathcal{M}M﻿ available in IR

  • Precision and recall
  • Since an experimental collection consists of many topics, the performance of an IR system is characterized by a set of scores, one for each topic
